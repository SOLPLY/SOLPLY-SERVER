       ON ps.place_id = p.id
WHERE p.town_id IN (301,302,303,304,305,306,307,308,309,310,311,312,313,314,315,316,317,318)

  AND EXISTS (SELECT 1 FROM place_tag pt
               WHERE pt.place_id = p.id AND pt.tag_id = 1)
  AND EXISTS (SELECT /*+ QB_NAME(qb1) */ 1 FROM place_tag pt
