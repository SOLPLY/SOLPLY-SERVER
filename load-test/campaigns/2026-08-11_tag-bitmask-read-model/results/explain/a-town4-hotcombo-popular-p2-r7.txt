SELECT ps.place_id, ps.popular_score, ps.bookmark_count,
       ps.review_count, ps.avg_rating
FROM place_stats ps
WHERE ps.town_id IN (301,302,303,304)
  AND ps.score_calculated_at IS NOT NULL

  AND EXISTS (SELECT 1 FROM place_tag pt
               WHERE pt.place_id = ps.place_id AND pt.tag_id = 1)
  AND EXISTS (SELECT 1 FROM place_tag pt
               WHERE pt.place_id = ps.place_id AND pt.tag_id IN (7,8,9,10))
  AND EXISTS (SELECT 1 FROM place_tag pt
               WHERE pt.place_id = ps.place_id AND pt.tag_id IN (11,12,13,14,15,16))
  AND (ps.popular_score < 7.551727E0
       OR (ps.popular_score = 7.551727E0 AND ps.place_id > 10133))
ORDER BY ps.popular_score DESC, ps.place_id ASC LIMIT 11
