       ps.review_count, ps.avg_rating
FROM place_stats ps
WHERE ps.town_id IN (301,302,303,304)

  AND (ps.tag_bitmask & 2) != 0
  AND (ps.tag_bitmask & 1920) != 0
  AND (ps.tag_bitmask & 129024) != 0
  AND (ps.created_at < '2026-03-24 01:27:15'
       OR (ps.created_at = '2026-03-24 01:27:15' AND ps.place_id < 10313))
ORDER BY ps.created_at DESC, ps.place_id DESC LIMIT 11
