SELECT  ps.place_id, ps.popular_score, ps.bookmark_count,
       ps.review_count, ps.avg_rating
FROM place_stats ps
WHERE ps.town_id IN (301,302,303,304,305,306,307,308,309,310,311,312,313,314,315,316,317,318)
  AND ps.score_calculated_at IS NOT NULL

  AND EXISTS (SELECT 1 FROM place_tag pt
               WHERE pt.place_id = ps.place_id AND pt.tag_id = 9003)
ORDER BY ps.popular_score DESC, ps.place_id ASC LIMIT 11
