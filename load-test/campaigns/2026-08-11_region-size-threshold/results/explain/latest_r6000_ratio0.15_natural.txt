FROM places p

       ON ps.place_id = p.id

  AND p.active = 1

  AND EXISTS (SELECT 1 FROM place_tag pt
               WHERE pt.place_id = p.id AND pt.tag_id = 9026)
ORDER BY p.created_at DESC, p.id DESC LIMIT 11
