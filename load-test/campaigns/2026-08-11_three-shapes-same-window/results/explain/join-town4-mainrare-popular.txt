--- SQL ---
SELECT ps.place_id, ps.popular_score, ps.bookmark_count,
       ps.review_count, ps.avg_rating
FROM place_stats ps
WHERE ps.town_id IN (301,302,303,304)
  AND ps.score_calculated_at IS NOT NULL
  AND EXISTS (SELECT 1 FROM place_tag pt JOIN tags t ON t.id = pt.tag_id
               WHERE pt.place_id = ps.place_id AND t.id = 5 AND t.active = 1)
ORDER BY ps.popular_score DESC, ps.place_id ASC LIMIT 11
